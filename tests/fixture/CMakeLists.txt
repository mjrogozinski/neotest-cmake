cmake_minimum_required(VERSION 3.21)
project(neotest_cmake_fixture LANGUAGES CXX)
enable_testing()
find_package(GTest CONFIG REQUIRED)
add_executable(cases cases.cpp cases_extra.cpp)
target_link_libraries(cases PRIVATE GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(cases TEST_PREFIX "discovered+[x]::")
add_executable(grouped grouped.cpp)
target_link_libraries(grouped PRIVATE GTest::gtest_main)
add_test(NAME "grouped+[all]" COMMAND grouped)
set_tests_properties("grouped+[all]" PROPERTIES ENVIRONMENT "ADAPTER_ENV=present" FIXTURES_REQUIRED prepared)
add_test(NAME prepare COMMAND "${CMAKE_COMMAND}" -E touch "${CMAKE_CURRENT_BINARY_DIR}/prepared")
set_tests_properties(prepare PROPERTIES FIXTURES_SETUP prepared)
add_test(NAME cleanup COMMAND "${CMAKE_COMMAND}" -E rm -f "${CMAKE_CURRENT_BINARY_DIR}/prepared")
set_tests_properties(cleanup PROPERTIES FIXTURES_CLEANUP prepared)
add_executable(policy policy.cpp)
target_link_libraries(policy PRIVATE GTest::gtest_main)
add_test(NAME expected_failure COMMAND policy --gtest_filter=Policy.ExpectedFailure)
set_tests_properties(expected_failure PROPERTIES WILL_FAIL TRUE)
add_test(NAME timeout COMMAND policy --gtest_filter=Policy.Timeout)
set_tests_properties(timeout PROPERTIES TIMEOUT 0.1)
add_test(NAME skip_rule COMMAND policy --gtest_filter=Policy.SkipRule)
set_tests_properties(skip_rule PROPERTIES SKIP_REGULAR_EXPRESSION SKIP_POLICY)
add_executable(unregistered unregistered.cpp)
target_link_libraries(unregistered PRIVATE GTest::gtest_main)
add_library(second_cases OBJECT second.cpp)
target_link_libraries(second_cases PRIVATE GTest::gtest)
add_executable(second_unregistered second_main.cpp)
target_link_libraries(second_unregistered PRIVATE second_cases GTest::gtest_main)
